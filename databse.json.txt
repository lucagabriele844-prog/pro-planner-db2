class MainHudScreen extends StatefulWidget {
  final String avatarName;
  final String calories;

  const MainHudScreen({super.key, required this.avatarName, required this.calories});

  @override
  State<MainHudScreen> createState() => _MainHudScreenState();
}

class _MainHudScreenState extends State<MainHudScreen> {
  int _currentTabIndex = 0;
  int _activeWeekIndex = 0;
  int _selectedDayIndex = 0;
  bool _isTrainingDayON = true;
  
  final List<String> _daysOfWeek = ["Lunedì", "Martedì", "Mercoledì", "Giovedì", "Venerdì", "Sabato", "Domenica"];
  late Future<List<WeekData>> _futureDatabase;

  @override
  void initState() {
    super.initState();
    _futureDatabase = fetchRemoteDatabase();
  }

  @override
  Widget build(BuildContext context) {
    return FutureBuilder<List<WeekData>>(
      future: _futureDatabase,
      builder: (context, snapshot) {
        if (snapshot.connectionState == ConnectionState.waiting) {
          return const Scaffold(
            body: Center(child: CircularProgressIndicator(color: AppColors.accentNeon)),
          );
        } else if (snapshot.hasError) {
          return Scaffold(
            body: Center(child: Text('Errore di connessione: ${snapshot.error}')),
          );
        } else if (!snapshot.hasData || snapshot.data!.isEmpty) {
          return const Scaffold(
            body: Center(child: Text('Nessun dato disponibile sul cloud.')),
          );
        }

        final database = snapshot.data!;
        return _buildActualHud(database);
      },
    );
  }

  Widget _buildActualHud(List<WeekData> database) {
    final activeWeek = database[_activeWeekIndex];
    final int baseCalories = int.tryParse(widget.calories) ?? 2800;
    final int adjustedCalories = _isTrainingDayON ? baseCalories : baseCalories - 400;

    return Scaffold(
      appBar: AppBar(
        backgroundColor: AppColors.surface,
        elevation: 0,
        leading: IconButton(
          icon: const Icon(Icons.switch_account, color: AppColors.accentNeon),
          tooltip: 'Cambia Profilo',
          onPressed: () => Navigator.pushReplacement(
            context,
            MaterialPageRoute(builder: (context) => const AvatarSelectionScreen()),
          ),
        ),
        title: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(widget.avatarName, style: const TextStyle(fontSize: 10, fontWeight: FontWeight.bold, color: AppColors.accentNeon, letterSpacing: 1.2)),
            Text('$adjustedCalories kcal Target (${_isTrainingDayON ? "ON Training" : "OFF Rest"})', style: const TextStyle(fontSize: 13, fontWeight: FontWeight.w700)),
          ],
        ),
        actions: [
          Padding(
            padding: const EdgeInsets.symmetric(vertical: 8.0, horizontal: 4.0),
            child: InkWell(
              onTap: () => setState(() => _isTrainingDayON = !_isTrainingDayON),
              borderRadius: BorderRadius.circular(10),
              child: Container(
                padding: const EdgeInsets.symmetric(horizontal: 10),
                decoration: BoxDecoration(
                  color: _isTrainingDayON ? AppColors.accentGreen.withValues(alpha: 0.2) : AppColors.accentOrange.withValues(alpha: 0.2),
                  borderRadius: BorderRadius.circular(10),
                  border: Border.all(color: _isTrainingDayON ? AppColors.accentGreen : AppColors.accentOrange, width: 1),
                ),
                child: Center(
                  child: Text(
                    _isTrainingDayON ? 'ON' : 'OFF',
                    style: TextStyle(
                      color: _isTrainingDayON ? AppColors.accentGreen : AppColors.accentOrange,
                      fontWeight: FontWeight.bold,
                      fontSize: 11,
                    ),
                  ),
                ),
              ),
            ),
          ),
          Padding(
            padding: const EdgeInsets.only(right: 12.0, left: 4.0),
            child: TextButton.icon(
              onPressed: () {
                setState(() {
                  _activeWeekIndex = (_activeWeekIndex + 1) % database.length;
                  _selectedDayIndex = 0;
                });
              },
              icon: const Icon(Icons.sync, size: 16, color: AppColors.accentNeon),
              label: Text('W${_activeWeekIndex + 1}/${database.length}', style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 12)),
              style: TextButton.styleFrom(
                foregroundColor: Colors.white,
                backgroundColor: AppColors.primary.withValues(alpha: 0.2),
                shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
              ),
            ),
          )
        ],
      ),
      body: _currentTabIndex == 0 ? _buildSettimanaTab(activeWeek) : _buildSpesaTab(activeWeek, database),
      bottomNavigationBar: Container(
        decoration: BoxDecoration(
          color: AppColors.surface,
          border: Border(top: BorderSide(color: Colors.white.withValues(alpha: 0.05))),
        ),
        child: NavigationBar(
          selectedIndex: _currentTabIndex,
          onDestinationSelected: (int index) => setState(() => _currentTabIndex = index),
          backgroundColor: Colors.transparent,
          indicatorColor: AppColors.primary.withValues(alpha: 0.3),
          destinations: const [
            NavigationDestination(icon: Icon(Icons.calendar_view_week_outlined), selectedIcon: Icon(Icons.calendar_view_week, color: AppColors.accentNeon), label: 'Programma'),
            NavigationDestination(icon: Icon(Icons.shopping_bag_outlined), selectedIcon: Icon(Icons.shopping_bag, color: AppColors.accentNeon), label: 'Spesa Unificata'),
          ],
        ),
      ),
    );
  }

  Widget _buildSettimanaTab(WeekData week) {
    final safeDayIndex = _selectedDayIndex.clamp(0, week.days.length - 1);
    final todayMeals = week.days[safeDayIndex].meals;

    return Column(
      children: [
        Container(
          padding: const EdgeInsets.symmetric(vertical: 12),
          decoration: BoxDecoration(
            color: AppColors.surface.withValues(alpha: 0.6),
            border: Border(bottom: BorderSide(color: Colors.white.withValues(alpha: 0.03))),
          ),
          child: SingleChildScrollView(
            scrollDirection: Axis.horizontal,
            padding: const EdgeInsets.symmetric(horizontal: 16),
            child: Row(
              children: List.generate(_daysOfWeek.length, (index) {
                bool isSelected = index == _selectedDayIndex;
                return GestureDetector(
                  onTap: () => setState(() => _selectedDayIndex = index),
                  child: AnimatedContainer(
                    duration: const Duration(milliseconds: 200),
                    margin: const EdgeInsets.only(right: 8),
                    padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
                    decoration: BoxDecoration(
                      gradient: isSelected
                          ? const LinearGradient(colors: [AppColors.primary, Color(0xFF1D4ED8)])
                          : null,
                      color: isSelected ? null : AppColors.background,
                      borderRadius: BorderRadius.circular(12),
                      border: Border.all(color: isSelected ? AppColors.accentNeon : Colors.white.withValues(alpha: 0.05)),
                    ),
                    child: Text(
                      _daysOfWeek[index],
                      style: TextStyle(
                        color: isSelected ? Colors.white : AppColors.textMuted,
                        fontWeight: isSelected ? FontWeight.bold : FontWeight.normal,
                        fontSize: 13,
                      ),
                    ),
                  ),
                );
              }),
            ),
          ),
        ),
        Expanded(
          child: ListView.builder(
            padding: const EdgeInsets.all(16),
            itemCount: todayMeals.length,
            itemBuilder: (context, index) {
              return MealCard(
                meal: todayMeals[index],
                isTrainingDay: _isTrainingDayON,
              );
            },
          ),
        ),
      ],
    );
  }

  Widget _buildSpesaTab(WeekData week, List<WeekData> database) {
    Set<String> uniqueIngredients = {};
    for (var day in week.days) {
      for (var meal in day.meals) {
        for (var ing in meal.ingredients) {
          // Rimuove grammature e millilitri tra parentesi (es. "(60g)", "(150ml)") per la lista spesa
          String cleanIng = ing.replaceAll(RegExp(r'\s*\([^)]*\)'), '').trim();
          if (cleanIng.isNotEmpty) {
            uniqueIngredients.add(cleanIng);
          }
        }
      }
    }
    List<String> spesaList = uniqueIngredients.toList()..sort();

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Container(
          width: double.infinity,
          padding: const EdgeInsets.all(16),
          decoration: BoxDecoration(
            color: AppColors.surface.withValues(alpha: 0.6),
            border: Border(bottom: BorderSide(color: Colors.white.withValues(alpha: 0.03))),
          ),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text('BLOCCO SPESA • W${_activeWeekIndex + 1} (MATERIE PRIME)', style: const TextStyle(color: AppColors.accentNeon, fontWeight: FontWeight.bold, fontSize: 10, letterSpacing: 1.5)),
              const SizedBox(height: 4),
              Text(week.focusTitle, style: const TextStyle(fontSize: 15, fontWeight: FontWeight.bold, color: Colors.white)),
            ],
          ),
        ),
        Expanded(
          child: ListView.builder(
            padding: const EdgeInsets.all(16),
            itemCount: spesaList.length,
            itemBuilder: (context, index) {
              return SpesaCheckbox(itemName: spesaList[index]);
            },
          ),
        ),
      ],
    );
  }
}

class MealCard extends StatelessWidget {
  final MealData meal;
  final bool isTrainingDay;

  const MealCard({super.key, required this.meal, required this.isTrainingDay});

  @override
  Widget build(BuildContext context) {
    final displayedIngredients = meal.ingredients.map((ing) {
      if (ing.contains("Pasta") || ing.contains("Riso") || ing.contains("Avena") || ing.contains("Pane") || ing.contains("Patate")) {
        return isTrainingDay ? "$ing (Carico High)" : "$ing (Ridotto Low)";
      }
      return ing;
    }).toList();

    return Container(
      margin: const EdgeInsets.only(bottom: 14),
      decoration: BoxDecoration(
        gradient: LinearGradient(
          colors: [AppColors.surface, AppColors.surfaceLight.withValues(alpha: 0.6)],
          begin: Alignment.topLeft,
          end: Alignment.bottomRight,
        ),
        borderRadius: BorderRadius.circular(18),
        border: Border.all(color: Colors.white.withValues(alpha: 0.04)),
      ),
      child: Padding(
        padding: const EdgeInsets.all(18.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Row(
              mainAxisAlignment: MainAxisAlignment.spaceBetween,
              children: [
                Text(
                  meal.slot.toUpperCase(),
                  style: const TextStyle(fontSize: 10, fontWeight: FontWeight.bold, color: AppColors.accentNeon, letterSpacing: 1.2),
                ),
                Container(
                  padding: const EdgeInsets.all(6),
                  decoration: BoxDecoration(
                    color: AppColors.background,
                    borderRadius: BorderRadius.circular(8),
                  ),
                  child: const Icon(Icons.restaurant, size: 12, color: AppColors.textMuted),
                ),
              ],
            ),
            const SizedBox(height: 10),
            Text(meal.title, style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold, color: Colors.white)),
            const SizedBox(height: 12),
            ...displayedIngredients.map((ing) => Padding(
              padding: const EdgeInsets.only(bottom: 5),
              child: Row(
                children: [
                  Container(
                    width: 4,
                    height: 4,
                    decoration: BoxDecoration(
                      color: isTrainingDay ? AppColors.accentGreen : AppColors.accentOrange,
                      shape: BoxShape.circle,
                    ),
                  ),
                  const SizedBox(width: 10),
                  Text(ing, style: TextStyle(color: AppColors.textMuted, fontSize: 13)),
                ],
              ),
            )),
          ],
        ),
      ),
    );
  }
}

class SpesaCheckbox extends StatefulWidget {
  final String itemName;
  const SpesaCheckbox({super.key, required this.itemName});

  @override
  State<SpesaCheckbox> createState() => _SpesaCheckboxState();
}

class _SpesaCheckboxState extends State<SpesaCheckbox> {
  bool _isChecked = false;

  @override
  Widget build(BuildContext context) {
    return AnimatedContainer(
      duration: const Duration(milliseconds: 200),
      margin: const EdgeInsets.only(bottom: 8),
      decoration: BoxDecoration(
        color: _isChecked ? AppColors.surface.withValues(alpha: 0.3) : AppColors.surface,
        borderRadius: BorderRadius.circular(14),
        border: Border.all(color: _isChecked ? Colors.transparent : Colors.white.withValues(alpha: 0.03)),
      ),
      child: CheckboxListTile(
        title: Text(
          widget.itemName,
          style: TextStyle(
            fontSize: 14,
            decoration: _isChecked ? TextDecoration.lineThrough : null,
            color: _isChecked ? AppColors.textMuted : Colors.white,
            fontWeight: _isChecked ? FontWeight.normal : FontWeight.w500,
          ),
        ),
        value: _isChecked,
        activeColor: AppColors.primary,
        checkColor: Colors.white,
        checkboxShape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(4)),
        onChanged: (bool? val) => setState(() => _isChecked = val ?? false),
        controlAffinity: ListTileControlAffinity.leading,
      ),
    );
  }
}

class MealData {
  final String slot;
  final String title;
  final List<String> ingredients;
  MealData(this.slot, this.title, this.ingredients);

  factory MealData.fromJson(Map<String, dynamic> json) {
    return MealData(
      json['slot'] ?? '',
      json['title'] ?? '',
      List<String>.from(json['ingredients'] ?? []),
    );
  }
}

class DayData {
  final List<MealData> meals;
  DayData(this.meals);

  factory DayData.fromJson(Map<String, dynamic> json) {
    var mealsList = json['meals'] as List;
    List<MealData> meals = mealsList.map((i) => MealData.fromJson(i)).toList();
    return DayData(meals);
  }
}

class WeekData {
  final String focusTitle;
  final List<DayData> days;
  WeekData(this.focusTitle, this.days);

  factory WeekData.fromJson(Map<String, dynamic> json) {
    var daysList = json['days'] as List;
    List<DayData> days = daysList.map((i) => DayData.fromJson(i)).toList();
    return WeekData(
      json['focusTitle'] ?? '',
      days,
    );
  }
}

Future<List<WeekData>> fetchRemoteDatabase() async {
  final response = await http.get(Uri.parse('https://raw.githubusercontent.com/lucagabriele844-prog/pro-planner-db2/refs/heads/main/databse.json.txt'));
  
  if (response.statusCode == 200) {
    List<dynamic> body = json.decode(response.body);
    return body.map((item) => WeekData.fromJson(item)).toList();
  } else {
    throw Exception('Impossibile caricare il piano nutrizionale dal cloud.');
  }
}
